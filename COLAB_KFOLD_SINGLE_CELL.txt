# ===== SAVE TO DRIVE =====
import pathlib, datetime, shutil, subprocess
from pathlib import Path
DATE=datetime.datetime.now().strftime("%Y%m%d_%H%M")
OUT=Path(f"/content/drive/MyDrive/kfold_{DATE}")
OUT.mkdir(parents=True, exist_ok=True)
for p in Path("kfold").glob("*"): 
    if p.is_file(): shutil.copy(p, OUT/p.name)
for p in Path("figures").glob("fig_kfold*"): shutil.copy(p, OUT/p.name)
for p in Path("results").glob("kfold*.json"): shutil.copy(p, OUT/p.name)
print(f"\n✅ Saved to {OUT}")
print(f"For paper: {OUT}/kfold_table.tex (Table: Mean±SD) + {OUT}/fig_kfold_box.png")
subprocess.run(["ls","-lh",str(OUT)], check=False)
