# ===== SAVE TO DRIVE/ablation with all results =====
import pathlib, datetime, shutil, subprocess
from pathlib import Path
DATE=datetime.datetime.now().strftime("%Y%m%d_%H%M")
OUT=Path(f"/content/drive/MyDrive/ablation_{DATE}")
OUT.mkdir(parents=True, exist_ok=True)
for p in Path("figures").glob("*"): shutil.copy(p, OUT/p.name)
for p in Path("ablation").glob("*"): 
    if p.is_file(): shutil.copy(p, OUT/p.name)
for p in Path("results").glob("*.json"): shutil.copy(p, OUT/p.name)
for p in Path("results").glob("*.tex"): shutil.copy(p, OUT/p.name)
print(f"✅ Saved to {OUT} — {len(list(Path('figures').glob('*.png')))} PNGs")
subprocess.run(["ls","-lh",str(OUT)], check=False)
print("Open Drive -> ablation_... for figures + ablation_table.tex")
