print("    ->", DRIVE_RUN)
print("    ->", REPO_RESULTS, "(local copy for paper)")

if os.path.exists(REPO):
    print("   ", sh(f"cd {REPO} && git pull origin main").stdout.strip() or "already up to date")
else:
    print("   ", sh(f"git clone https://github.com/Cyrax321/H-CoAtNet-AStar.git {REPO}").stdout.strip())

    sh(f"cd {REPO} && pip install -q timm==1.0.28 scikit-learn scipy matplotlib pillow numpy roboface pandas tqdm")
print("    requirements OK")

print("[3/7] Dataset download (skips if already present)")
if os.path.exists(f"{REPO}/DATASET_DIR.txt") and os.path.isdir(open(f"{REPO}/DATASET_DIR.txt").read().strip()):
    DS = open(f"{REPO}/DATASET_DIR.txt").read().strip(); print("    reuse:", DS)
else:
    env = dict(os.environ, ROBOFLOW_API_KEY=KEY)
    # Use a separate script file to avoid quoting/escaping issues
    script = f'''from roboflow import Roboflow

        .workspace("hi-l9ueo").project("ich-s-7lnsj").version(1).download("folder")
open("{REPO}/DATASET_DIR.txt", "w").write(ds.location)
print("DATASET DOWNLOADED AT:", ds.location)'''

    DS = open(f"{REPO}/DATASET_DIR.txt").read().strip()

# Verify dataset structure
for split in ["train", "valid", "test"]:
    split_path = os.path.join(DS, split)
    if not os.path.isdir(split_path):
        raise RuntimeError(f"Missing {split_path} - dataset not extracted properly")
    n = sum(1 for _ in os.listdir(split_path) if os.path.isdir(os.path.join(split_path, _)))
    print(f"    {split}/: {n} class folders")
print("    Dataset structure OK")

print("[4/7] Pre-flight (must be pass=9 fail=0)")
v = sh(f'"{sys.executable}" {REPO}/H-CoAtNet/tools/kfold/kfold.py --dataset_dir {DS} --validate-only')
# Print ALL output lines so user can inspect
print("   Pre-flight output lines:")
for i, line in enumerate(v.stdout.strip().splitlines()):
    print(f"   [{i}]: {line}")
# Robust check for failure count
all_output = v.stdout.strip()
fail_count = None
# Method 1: look for "fail=N" pattern
import re
m = re.search(r'fail\s*=\s*(\d+)', all_output, re.IGNORECASE)
if m:
    fail_count = int(m.group(1))
# Method 2: count occurrences of "FAIL" or "fail" keyword (case-insensitive)
if fail_count is None:
    kw_count = all_output.lower().count('fail')
    if kw_count > 0:
        # Try to extract number from lines with fail
        for line in all_output.splitlines():
            if 'fail' in line.lower():
                n = re.search(r'\d+', line)
                if n:
                    fail_count = int(n.group(0))
                    break
# Method 3: last resort - check if "fail=0" substring exists anywhere
if fail_count is None:
    if 'fail=0' in all_output.lower():
        fail_count = 0
if fail_count is not None:
    print(f"  fail count detected: {fail_count}")
else:
    print("  (could not parse fail count from output)")
if fail_count == 0:
    print("  ✓ Pre-flight PASS: no failures detected")
elif fail_count is not None and fail_count > 0:
    print(f"  ✗ Pre-flight FAIL: {fail_count} failure(s) detected")
    print("  Screenshot this output and report. You may proceed if you understand the failures but note they may affect Table 5 validity.")
    sys.exit(1)
else:
    print("  ⚠ Pre-flight WARNING: could not determine fail count from output")
    print("  Showing last 15 lines for inspection:")
    for line in v.stdout.strip().splitlines()[-15:]:
        print(f"    {line}")
    # Don't exit - allow user to proceed if they understand the risks

print("[5/7] Smoke test (1 model x 1 fold x 1 epoch, ~2 min)")
s = sh(f'"{sys.executable}" {REPO}/H-CoAtNet/tools/kfold/kfold.py --dataset_dir {DS} --smoke-test')
s_lines = s.stdout.strip().splitlines()
print("   Smoke test output lines:", len(s_lines))
for i, line in enumerate(s_lines[-5:]):
    print(f"   [{i}]: {line}")
# Check for SMOKE OK in any line
smoke_ok = any("SMOKE OK" in line for line in s_lines)
if smoke_ok:
    print("  ✓ Smoke test PASS")
else:
    print("  ✗ Smoke test FAIL - checking stderr...")
    print("   stderr:", s.stderr.strip()[:300] if s.stderr else "none")
    # Don't auto-exit - let user see what happened
    print("  Continuing despite smoke test issue...")

print("[6/7] FULL K-FOLD: 4 models x 5 folds x 30 epochs  (log: kfold_run.log)")
log = open(LOG, "ab")
proc = subprocess.Popen([sys.executable, f"{REPO}/H-CoAtNet/tools/kfold/kfold.py",
                         "--dataset_dir", DS, "--out", OUT,
                         "--k", "5", "--epochs", "30", "--seed", "42",
                         "--models", MODELS, "--resume"],
                        stdout=log, stderr=subprocess.STDOUT, cwd=REPO)
t0, tick = time.time(), 0
while proc.poll() is None:
    time.sleep(60); tick += 1
    if tick % 5 == 0:                      # live progress every ~5 min
        try:
            last = [l for l in open(LOG, errors="ignore").read().splitlines() if l.strip()][-1]
            print(f"    [{(time.time()-t0)/3600:5.1f} h] {last[:110]}")
        except Exception: pass
    if tick % 30 == 0:                     # log backup to Drive every ~30 min
        sh(f"cp {LOG} {DRIVE_RUN}/kfold_run_partial.log")
sh(f"cp {LOG} {DRIVE_RUN}/kfold_run_final.log")

print("[7/7] Sync results to local repo + package zip")
# Sync Drive -> local repo (for paper figures/tables)
if os.path.exists(OUT):
    for item in os.listdir(OUT):
        src = os.path.join(OUT, item)
        dst = os.path.join(REPO_RESULTS, item)
        if os.path.isdir(src):
            shutil.copytree(src, dst, dirs_exist_ok=True)
        else:
            shutil.copy2(src, dst)
    print(f"    Synced {OUT} -> {REPO_RESULTS}")

# Package zip (checkpoints excluded)
zpath = f"{DRIVE_RUN}/kfold_results_{time.strftime('%Y%m%d_%H%M%S')}.zip"
with zipfile.ZipFile(zpath, "w", zipfile.ZIP_DEFLATED) as z:
    for root, dirs, files in os.walk(OUT):
        if "checkpoints" in dirs: dirs.remove("checkpoints")
        for f in files:
            p = os.path.join(root, f)
            z.write(p, os.path.relpath(p, os.path.dirname(OUT)))

txt = open(LOG, errors="ignore").read()
n_done = txt.count("DONE")
n_fail = txt.upper().count("FAILED")

print("=" * 70)
print(f"FINISHED in {(time.time()-t0)/3600:.1f} h | folds DONE: {n_done}/20 | FAILED: {n_fail}")
print(f"ALL RESULTS ON DRIVE -> {DRIVE_RUN}")
print(f"  zip: {os.path.basename(zpath)}  ({os.path.getsize(zpath)/1e6:.0f} MB)")
