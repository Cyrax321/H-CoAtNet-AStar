(key embedded), pre-flight, 7-model x 5-fold training, results saved to Drive,

HOW TO USE:
1. colab.research.google.com -> New notebook
2. Runtime -> Change runtime type -> T4 GPU  (IMPORTANT)
3. Paste the cell below into the notebook and press Run
4. Authorize Google Drive when asked
5. Pre-flight MUST print: VALIDATION SUMMARY: pass=9 warn=0 fail=0
   (any FAIL -> stop and screenshot the output)
6. Keep the tab open ~8-12 h. Progress prints every ~5 min.
7. If Colab disconnects overnight: new session -> T4 -> paste the SAME cell
   -> Run again. It resumes from the last completed fold (nothing is redone).
8. Morning check: last line prints  folds DONE: 35/35 | FAILED: 0
   Then take everything from Drive: MyDrive/HCoAtNet_KFOLD_RUN/
   - kfold_results_<stamp>.zip   (all CSVs, Table 6 LaTeX, 13 figures, stats)

# H-CoAtNet K-FOLD — ALL-IN-ONE COLAB CELL (T4)          [R2-4, seed 42]
# BEFORE RUNNING: Runtime -> Change runtime type -> T4 GPU
# Keep this tab open while it runs (~8-12 h).
# If Colab disconnects: re-run THIS SAME CELL -> it resumes automatically.
# =============================================================================
import os, subprocess, sys, time, zipfile, glob

KEY        = "gXuxxWEMFJ8nK73o7pN7"
REPO       = "/content/H-CoAtNet-AStar"
DRIVE_RUN  = "/content/drive/MyDrive/HCoAtNet_KFOLD_RUN"
OUT        = f"{DRIVE_RUN}/results/kfold"          # results live ON Drive -> resume-safe
LOG        = "/content/kfold_run.log"

def sh(cmd, **kw):
    return subprocess.run(cmd, shell=True, text=True, capture_output=True, **kw)

print("=" * 70)
print("[0/6] GPU check")
r = sh("nvidia-smi --query-gpu=name,memory.total --format=csv,noheader")
print("   ", r.stdout.strip() if r.returncode == 0 else "NO GPU! Runtime -> T4 GPU, then re-run."); assert r.returncode == 0

print("[1/6] Mounting Google Drive")
from google.colab import drive
if not os.path.ismount("/content/drive"):
    drive.mount("/content/drive")
os.makedirs(DRIVE_RUN, exist_ok=True)
print("    ->", DRIVE_RUN)

print("[2/6] Code + requirements")
if os.path.exists(REPO):
    print("   ", sh(f"cd {REPO} && git pull origin main").stdout.strip() or "already up to date")
else:
    print("   ", sh(f"git clone https://github.com/Cyrax321/H-CoAtNet-AStar.git {REPO}").stdout.strip())
i = sh(f"cd {REPO} && pip install -q -r requirements.txt")
if i.returncode != 0:
    sh(f"cd {REPO} && pip install -q torch torchvision timm scikit-learn scipy matplotlib pillow numpy roboflow pandas tqdm")
print("    requirements OK")

print("[3/6] Dataset download (skips if already present)")
if os.path.exists(f"{REPO}/DATASET_DIR.txt") and os.path.isdir(open(f"{REPO}/DATASET_DIR.txt").read().strip()):
    DS = open(f"{REPO}/DATASET_DIR.txt").read().strip(); print("    reuse:", DS)
else:
    env = dict(os.environ, ROBOFLOW_API_KEY=KEY)
    code = ('from roboflow import Roboflow\n'
            'ds = Roboflow(api_key=__import__("os").environ["ROBOFLOW_API_KEY"])'
            '.workspace("hi-l9ueo").project("ich-s-7lnsj").version(1).download("folder")\n'
            f'open("{REPO}/DATASET_DIR.txt","w").write(ds.location)\n'
            'print("DATASET DOWNLOADED AT:", ds.location)')
    r = sh(f'"{sys.executable}" -c {code!r}', env=env)
    print("   ", r.stdout.strip() or r.stderr[-400:]); assert r.returncode == 0
    DS = open(f"{REPO}/DATASET_DIR.txt").read().strip()

print("[4/6] Pre-flight (must be pass=9 fail=0)")
v = sh(f'"{sys.executable}" {REPO}/H-CoAtNet/tools/kfold/kfold.py --dataset_dir {DS} --validate-only')
print("\n".join(v.stdout.strip().splitlines()[-12:]))
assert "fail=0" in v.stdout, "PRE-FLIGHT FAILED - screenshot this output"

print("[5/6] FULL K-FOLD: 7 models x 5 folds x 30 epochs  (log: kfold_run.log)")
log = open(LOG, "ab")
proc = subprocess.Popen([sys.executable, f"{REPO}/H-CoAtNet/tools/kfold/kfold.py",
                         "--dataset_dir", DS, "--out", OUT,
                         "--k", "5", "--epochs", "30", "--seed", "42",
                         "--models", "all", "--resume"],
                        stdout=log, stderr=subprocess.STDOUT, cwd=REPO)
t0, tick = time.time(), 0
while proc.poll() is None:
    time.sleep(60); tick += 1
    if tick % 5 == 0:                      # live progress every ~5 min
        try:
            last = [l for l in open(LOG, errors="ignore").read().splitlines() if l.strip()][-1]
            print(f"    [{(time.time()-t0)/3600:5.1f} h] {last[:110]}")
        except Exception: pass
    if tick % 30 == 0:                     # log backup to Drive every ~30 min
        sh(f"cp {LOG} {DRIVE_RUN}/kfold_run_partial.log")
sh(f"cp {LOG} {DRIVE_RUN}/kfold_run_final.log")
txt = open(LOG, errors="ignore").read()
n_done, n_fail = txt.count("DONE"), txt.upper().count("FAILED")

print("[6/6] Packaging zip (checkpoints excluded)")
zpath = f"{DRIVE_RUN}/kfold_results_{time.strftime('%Y%m%d_%H%M%S')}.zip"
with zipfile.ZipFile(zpath, "w", zipfile.ZIP_DEFLATED) as z:
    for root, dirs, files in os.walk(OUT):
        if "checkpoints" in dirs: dirs.remove("checkpoints")
        for f in files:
            p = os.path.join(root, f)
            z.write(p, os.path.relpath(p, os.path.dirname(OUT)))
print("=" * 70)
print(f"FINISHED in {(time.time()-t0)/3600:.1f} h | folds DONE: {n_done}/35 | FAILED: {n_fail}")
print(f"ALL RESULTS ON DRIVE -> {DRIVE_RUN}")
print(f"  zip: {os.path.basename(zpath)}  ({os.path.getsize(zpath)/1e6:.0f} MB)")
print("  top files: kfold_test_summary.csv, kfold_summary.csv, kfold_table.tex,")
print("             kfold_per_class_summary.csv, kfold_friedman_nemenyi.json, figures/")
print("=" * 70)
