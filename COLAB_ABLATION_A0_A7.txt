print("=== STEP 5: full A0-A7 ablation (sequential, ~7 hours) ===")
print("Each variant: 30 epochs, ~50 min on T4. Backed up to Drive after each.")
VARIANTS = ["cnnOnly", "vitOnly", "seNoPrune", "randomPrune",
            "directL2", "hierarchical", "hierarchicalRaw", "noViT"]
overall_start = time.time()
for v in VARIANTS:
    oj_local = f"results/results_ablation_{v}.json"
    oj_drive = f"{RESULTS}/results_ablation_{v}.json"
    if os.path.exists(oj_local) or os.path.exists(oj_drive):
        print(f"SKIP {v}: already trained")
        continue
    t0 = time.time()
    log_path = f"{DRIVE}/logs/train_{v}.log"
    cmd = ["python3", "-u", "ablation study/ablation_study.py",
           "--variant", v, "--epochs", "30", "--seed", "42",
           "--dataset_dir", "/content/dataset"]
    print(f"\n[TRAIN {v}] started at {time.strftime('%H:%M:%S')}")
    # Stream output to BOTH the log file AND the cell output so epochs are visible live
    last_heartbeat = time.time()
    with open(log_path, "w") as logf:
        proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1)
        for line in proc.stdout:
            sys.stdout.write(line)
            sys.stdout.flush()
            logf.write(line)
            logf.flush()
            # Heartbeat: print elapsed time every 60s of wall clock so you
            # can confirm the cell is alive even between epoch prints.
            if time.time() - last_heartbeat > 60:
                last_heartbeat = time.time()
                elapsed_min = (time.time() - t0) / 60
                hb = f"  [heartbeat {v}] still running, {elapsed_min:.1f} min elapsed\n"
                sys.stdout.write(hb)
                sys.stdout.flush()
                logf.write(hb)
                logf.flush()
        proc.wait()
    print(f"\n[TRAIN {v}] done in {(time.time()-t0)/60:.1f} min (exit={proc.returncode})")
    if proc.returncode == 0:
        for src in [oj_local,
                    f"ablation study/results/results_ablation_{v}.json",
                    f"histories/history_ablation_{v}.json"]:
            if os.path.exists(src):
                shutil.copy(src, RESULTS)
        pth = f"ablation study/results/best_{v}.pth"
        if os.path.exists(pth):
            shutil.copy(pth, RESULTS)
        # Also backup per-variant figures immediately so we don't lose
        # them if a later variant crashes or Colab disconnects.
        for fig_ext in ("png",):
            for fig_pattern in (f"curve_{v}_*.{fig_ext}",
                                f"confusion_{v}_*.{fig_ext}"):
                for f in Path("ablation study/figures").glob(fig_pattern):
                    shutil.copy(f, f"{DRIVE}/figures/")
        print(f"[BACKUP] {v} -> {RESULTS}")
    else:
        # exit=3 is the ablation script's compare-failure (expected mid-run);
        # any other non-zero exit is a real crash worth stopping on.
        if proc.returncode == 3:
            print(f"[NOTE] {v} compare-after-train raised RuntimeError (expected; STEP 6 will do the final compare)")
        else:
            print(f"[FAIL] {v} exit={proc.returncode}; see {log_path}")
            break
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
print(f"\nAll variants done in {(time.time()-overall_start)/60:.1f} min total")

print()
