import subprocess, os, sys, time, shutil
from pathlib import Path

print("=== STEP 1: env + clone + install ===")
os.makedirs("/content", exist_ok=True)
os.chdir("/content")
subprocess.run(["rm", "-rf", "H-CoAtNet-AStar"], check=False)
subprocess.run(["git", "clone", "--depth", "1",
                "https://github.com/Cyrax321/H-CoAtNet-AStar.git"], check=True)
os.chdir("/content/H-CoAtNet-AStar")
subprocess.run(["git", "log", "--oneline", "-3"], check=False)

subprocess.run(["pip", "install", "-q", "-r", "requirements-colab.txt"], check=False)

import timm, torch
print("timm:", timm.__version__, "| torch:", torch.__version__,
      "| cuda:", torch.cuda.is_available())

os.environ["ROBOFLOW_API_KEY"] = "gXuxxWEMFJ8nK73o7pN7"

print()
print("=== STEP 2: Drive mount ===")
from google.colab import drive
drive.mount("/content/drive")
DRIVE = "/content/drive/MyDrive/HCoAtNet_ablation"
for sub in ["results", "figures", "histories", "logs"]:
    os.makedirs(f"{DRIVE}/{sub}", exist_ok=True)
RESULTS = f"{DRIVE}/results"

print()
print("=== STEP 3: pre-download IN1K weights + dataset (reused across reconnects) ===")
os.environ.setdefault("HF_HOME", "/root/.cache/huggingface")
print("Downloading convnext_tiny IN1K...")
_t = timm.create_model("convnext_tiny", pretrained=True, num_classes=0)
print("IN1K cached")
del _t
import gc; gc.collect()
if torch.cuda.is_available():
    torch.cuda.empty_cache()

# Pre-download Roboflow dataset once, into a STABLE location (not inside the
# cloned repo). STEP 1 does rm -rf H-CoAtNet-AStar on every reconnect, so the
# dataset would otherwise re-download each time.
STABLE_DATA = "/content/dataset"
if not os.path.exists(os.path.join(STABLE_DATA, "train")):
    print("Downloading Roboflow dataset once...")
    from roboflow import Roboflow
    rf = Roboflow(api_key=os.environ["ROBOFLOW_API_KEY"])
    ds = rf.workspace("hi-l9ueo").project("ich-s-7lnsj").version(1).download("folder")
    # Move to stable location
    import shutil as _sh
    if os.path.exists(STABLE_DATA):
        _sh.rmtree(STABLE_DATA)
    _sh.move(ds.location, STABLE_DATA)
    print(f"dataset cached at {STABLE_DATA}")
else:
    print(f"dataset already cached at {STABLE_DATA}")

print()
print("=== STEP 4: smoke test (1 epoch vitOnly, ~3 min) ===")
with open(f"{DRIVE}/logs/smoke.log", "w") as logf:
    p = subprocess.run(["python3", "-u", "ablation study/ablation_study.py",
                        "--variant", "smoke", "--epochs", "1", "--seed", "42",
                        "--dataset_dir", "/content/dataset"],
                       stdout=logf, stderr=subprocess.STDOUT)
print(f"smoke exit code: {p.returncode}")
with open(f"{DRIVE}/logs/smoke.log") as f:
    for line in f.read().splitlines()[-12:]:
        print(line)

print()
print("=== STEP 5: full A0-A7 ablation (sequential, ~7 hours) ===")
print("Each variant: 30 epochs, ~50 min on T4. Backed up to Drive after each.")
VARIANTS = ["cnnOnly", "vitOnly", "seNoPrune", "randomPrune",
            "directL2", "hierarchical", "hierarchicalRaw", "noViT"]
overall_start = time.time()
for v in VARIANTS:
    oj_local = f"results/results_ablation_{v}.json"
    oj_drive = f"{RESULTS}/results_ablation_{v}.json"
    # Resume on reconnect: skip variants already trained.
    # If local is missing but Drive has it (fresh clone after disconnect),
    # restore local from Drive so compare/zip can find it later.
    if os.path.exists(oj_drive) and not os.path.exists(oj_local):
        os.makedirs(os.path.dirname(oj_local), exist_ok=True)
        shutil.copy(oj_drive, oj_local)
        h_local = f"histories/history_ablation_{v}.json"
        h_drive = f"{RESULTS}/history_ablation_{v}.json"
        if os.path.exists(h_drive) and not os.path.exists(h_local):
            os.makedirs(os.path.dirname(h_local), exist_ok=True)
            shutil.copy(h_drive, h_local)
    if os.path.exists(oj_local) or os.path.exists(oj_drive):
        print(f"SKIP {v}: already trained")
        continue
    t0 = time.time()
    log_path = f"{DRIVE}/logs/train_{v}.log"
    cmd = ["python3", "-u", "ablation study/ablation_study.py",
           "--variant", v, "--epochs", "30", "--seed", "42",
           "--dataset_dir", "/content/dataset"]
    print(f"\n[TRAIN {v}] started at {time.strftime('%H:%M:%S')}")
    # Stream output to BOTH the log file AND the cell output so epochs are visible live
    last_heartbeat = time.time()
    with open(log_path, "w") as logf:
        proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1)
        for line in proc.stdout:
            sys.stdout.write(line)
            sys.stdout.flush()
            logf.write(line)
            logf.flush()
            # Heartbeat: print elapsed time every 60s of wall clock so you
            # can confirm the cell is alive even between epoch prints.
            if time.time() - last_heartbeat > 60:
                last_heartbeat = time.time()
                elapsed_min = (time.time() - t0) / 60
                hb = f"  [heartbeat {v}] still running, {elapsed_min:.1f} min elapsed\n"
                sys.stdout.write(hb)
                sys.stdout.flush()
                logf.write(hb)
                logf.flush()
        proc.wait()
    print(f"\n[TRAIN {v}] done in {(time.time()-t0)/60:.1f} min (exit={proc.returncode})")
    if proc.returncode == 0:
        for src in [oj_local,
                    f"ablation study/results/results_ablation_{v}.json",
                    f"histories/history_ablation_{v}.json"]:
            if os.path.exists(src):
                shutil.copy(src, RESULTS)
        pth = f"ablation study/results/best_{v}.pth"
        if os.path.exists(pth):
            shutil.copy(pth, RESULTS)
        # Also backup per-variant figures immediately so we don't lose
        # them if a later variant crashes or Colab disconnects.
        for fig_ext in ("png",):
            for fig_pattern in (f"curve_{v}_*.{fig_ext}",
                                f"confusion_{v}_*.{fig_ext}"):
                for f in Path("ablation study/figures").glob(fig_pattern):
                    shutil.copy(f, f"{DRIVE}/figures/")
        print(f"[BACKUP] {v} -> {RESULTS}")
    else:
        # exit=3 is the ablation script's compare-failure (expected mid-run);
        # any other non-zero exit is a real crash worth stopping on.
        if proc.returncode == 3:
            print(f"[NOTE] {v} compare-after-train raised RuntimeError (expected; STEP 6 will do the final compare)")
        else:
            print(f"[FAIL] {v} exit={proc.returncode}; see {log_path}")
            break
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
print(f"\nAll variants done in {(time.time()-overall_start)/60:.1f} min total")

print()
print("=== STEP 6: Table 5 + compare ===")
p = subprocess.run(["python3", "ablation study/ablation_study.py",
                    "--variant", "compare"], capture_output=True, text=True)
print("\n".join(p.stdout.splitlines()[-20:]))

print("\n--- TABLE 5 (LaTeX) ---")

    print(f.read())

    print(f.read())

print()
print("=== STEP 7: Drive backup (figures + zip) ===")
DRIVE_FIGS = f"{DRIVE}/figures"
os.makedirs(DRIVE_FIGS, exist_ok=True)
n_copied = 0
for ext in ("png", "pdf"):
    for f in Path("ablation study/figures").glob(f"*.{ext}"):
        shutil.copy(f, DRIVE_FIGS)
        n_copied += 1
for fname in ["ablation study/ablation_table.tex", "ablation study/ablation_summary.csv"]:
    if os.path.exists(fname):
        shutil.copy(fname, DRIVE)
print(f"copied {n_copied} figure files to {DRIVE_FIGS}")
