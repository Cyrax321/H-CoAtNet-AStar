import os, subprocess, time, sys, shutil
import torch

DRIVE = "/content/drive/MyDrive/HCoAtNet_ablation"
RESULTS = f"{DRIVE}/results"
os.makedirs(RESULTS, exist_ok=True)

VARIANTS = ["cnnOnly", "vitOnly", "seNoPrune", "randomPrune",
            "directL2", "hierarchical", "hierarchicalRaw", "noViT"]

print(f"Starting full A0-A7 ablation at {time.strftime('%Y-%m-%d %H:%M:%S')}")
overall_start = time.time()

for v in VARIANTS:
    out_json = f"results/results_ablation_{v}.json"
    out_json_drive = f"{RESULTS}/results_ablation_{v}.json"

    # Skip if already trained (resume on reconnect)
    if os.path.exists(out_json) or os.path.exists(out_json_drive):
        print(f"SKIP {v}: already trained ({out_json_drive} exists)")
        continue

    print(f"\n===== TRAIN {v} (30 epochs) =====")
    t0 = time.time()
    log_path = f"{DRIVE}/logs/train_{v}.log"
    cmd = ["python3", "-u", "ablation study/ablation_study.py",
           "--variant", v, "--epochs", "30", "--seed", "42"]
    with open(log_path, "w") as logf:
        proc = subprocess.run(cmd, stdout=logf, stderr=subprocess.STDOUT)
    elapsed = (time.time() - t0) / 60
    print(f"{v} done in {elapsed:.1f} min (exit={proc.returncode})")

    if proc.returncode == 0:
        for src in [out_json,
                    f"ablation study/results/results_ablation_{v}.json",
                    f"histories/history_ablation_{v}.json"]:
            if os.path.exists(src):
                shutil.copy(src, RESULTS)
        pth = f"ablation study/results/best_{v}.pth"
        if os.path.exists(pth):
            shutil.copy(pth, RESULTS)
        print(f"{v}: backed up to {RESULTS}")
    else:
        print(f"{v}: FAILED (exit={proc.returncode}); check {log_path}")
        break

    import gc; gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()

overall_min = (time.time() - overall_start) / 60
print(f"\nAll variants done in {overall_min:.1f} min total")
print("CELL 3 DONE")
