# ===== SAVE TO DRIVE — REAL ablation with all results =====
import pathlib, datetime, shutil, subprocess
from pathlib import Path
DATE=datetime.datetime.now().strftime("%Y%m%d_%H%M")
OUT=Path(f"/content/drive/MyDrive/ablation_REAL_{DATE}")
OUT.mkdir(parents=True, exist_ok=True)
for p in Path("figures").glob("*"): shutil.copy(p, OUT/p.name)
for p in Path("ablation").glob("*"): 
    if p.is_file(): shutil.copy(p, OUT/p.name)
for p in Path("results").glob("*.json"): shutil.copy(p, OUT/p.name)
for p in Path("results").glob("*.tex"): shutil.copy(p, OUT/p.name)
for p in Path("histories").glob("*.json"): 
    (OUT/"histories").mkdir(exist_ok=True); shutil.copy(p, OUT/"histories"/p.name)
print(f"\n✅ REAL ablation saved to {OUT}")
print(f"For paper: {OUT}/ablation_table.tex (Table 5) + {OUT}/fig_ablation_main.png (Fig.5)")
subprocess.run(["ls","-lh",str(OUT)], check=False)
