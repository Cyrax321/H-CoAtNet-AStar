import json
for v in ["full", "noSE", "noViT", "cnnOnly"]:
    d = json.load(open(f"results/results_ablation_{v}.json"))
    t = d["test"]
    print(v, "acc", round(t["accuracy"]*100,2), "bal", round(t["balanced_accuracy"]*100,2),
          "macroF1", round(t["macro"]["f1"]*100,2), "kappa", round(t["kappa"]*100,2),
          "mcc", round(t["mcc"]*100,2), "auroc", round((t.get("auroc_macro") or 0)*100,2),
          "ece", round(t["ece"]*100,2), "n", t["n"])
