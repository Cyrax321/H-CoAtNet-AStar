STEP 5 — FULL OVERNIGHT RUN (nohup keeps it alive after disconnect):
--------------------------------------------------------------------------------

    --dataset_dir <DATASET_DIR> \
    --k 5 --epochs 30 --seed 42 --models all --resume \
    > kfold_run.log 2>&1 &

# Monitor progress any time:

# If the box disconnects: log back in and re-run the SAME Step-5 command.
# Completed folds are skipped automatically (--resume).

STEP 5b — AUTO-BACKUP WATCHER (start right after Step 5; runs unattended):
--------------------------------------------------------------------------------
# Waits for training to finish, then zips results/kfold automatically
# (checkpoints excluded to keep the archive small; ~50-100 MB).
nohup bash -c '
  while kill -0 $(cat kfold_train.pid) 2>/dev/null; do sleep 120; done
  STAMP=$(date +%Y%m%d_%H%M%S)
  ZIP="kfold_results_${STAMP}.zip"
  zip -rq "$ZIP" results/kfold -x "results/kfold/checkpoints/*" \
      && echo "[BACKUP] saved $ZIP (checkpoints excluded)" >> kfold_run.log \
      || echo "[BACKUP] zip failed - run Step 6 manually" >> kfold_run.log
' > kfold_backup.log 2>&1 &
