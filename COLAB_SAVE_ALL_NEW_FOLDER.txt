# 2. Copy helper, skips missing, never deletes source
def copy_file(src, dst):
    src = Path(src)
    dst = Path(dst)
    if src.exists():
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dst)
        return True
    else:
        print(f"skip missing: {src}")
        return False

def copy_glob(pattern, dest_dir):
    import glob
    files = sorted(glob.glob(pattern, recursive=True))
    n = 0
    for f in files:
        if Path(f).is_file():
            copy_file(f, Path(dest_dir) / Path(f).name)
            n += 1
    print(f"{pattern} -> {n} files to {dest_dir}")
    return n

def copy_tree(src, dst):
    src = Path(src)
    dst = Path(dst)
    if src.exists():
        shutil.copytree(src, dst, dirs_exist_ok=True)
        n = sum(1 for _ in dst.rglob("*") if _.is_file())
        print(f"{src} -> {dst} ({n} files)")
    else:
        print(f"skip missing dir: {src}")
